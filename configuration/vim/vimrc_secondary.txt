" =========================
" Core Vim Behavior
" =========================
set nocompatible
set encoding=utf-8

" Use system clipboard (VS Code already does this, but safe)
set clipboard=unnamedplus

" Better backspace behavior
set backspace=indent,eol,start

" =========================
" UI / Visuals (VS Code–ish)
" =========================
set number
set relativenumber

set cursorline
set signcolumn=yes

set nowrap
set scrolloff=5
set sidescrolloff=5

set termguicolors
set background=dark

" =========================
" Indentation (VS Code-like)
" =========================
set expandtab
set tabstop=4
set shiftwidth=4
set softtabstop=4
set smartindent
set autoindent

" =========================
" Search Behavior
" =========================
set incsearch
set hlsearch
set ignorecase
set smartcase

nnoremap <Esc> :nohlsearch<CR>

" =========================
" Faster / Smoother
" =========================
set updatetime=300
set timeoutlen=500

" =========================
" VS Code Friendly Keybinds
" =========================

" Save like Ctrl+S
nnoremap <C-s> :w<CR>
inoremap <C-s> <Esc>:w<CR>a

" Undo / Redo feel natural
nnoremap U <C-r>

" Better line movement
nnoremap J mzJ`z

" Keep selection when indenting
vnoremap < <gv
vnoremap > >gv

" =========================
" Window / Pane Navigation
" =========================
nnoremap <C-h> <C-w>h
nnoremap <C-j> <C-w>j
nnoremap <C-k> <C-w>k
nnoremap <C-l> <C-w>l

" =========================
" Quality-of-Life
" =========================

" Yank to end of line like VS Code
nnoremap Y y$

" Make x not yank
nnoremap x "_x

" =========================
" VSCodeVim Specific
" =========================
if exists('g:vscode')
  " Let VS Code handle tabs, files, etc.
  set noshowmode
endif
